📊 بنية الفاتورة:
═══════════════════════════════════════════════════════════════════════════════
{
    id: timestamp,
    invoiceNumber: "string",
    date: "YYYY-MM-DD",
    client: {
        name: "string",
        phone: "string",
        email: "string",
        address: "string"
    },
    taxNumber: "string",
    taxRate: number,
    discountRate: number,
    feesRate: number,
    items: [
        { name, quantity, price, total }
    ],
    subtotal: number,
    discount: number,
    afterDiscount: number,
    fees: number,
    tax: number,
    total: number,
    notes: "string",
    createdAt: "string"
}

🧮 الحسابات الرياضية:
═══════════════════════════════════════════════════════════════════════════════
Subtotal = Σ(quantity × price) لكل بند
Discount = Subtotal × (DiscountRate / 100)
AfterDiscount = Subtotal - Discount
Fees = AfterDiscount × (FeesRate / 100)
Tax = AfterDiscount × (TaxRate / 100)
Total = AfterDiscount + Fees + Tax
